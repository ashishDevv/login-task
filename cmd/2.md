🟢 Middleware 1 (FIRST): Logger Injection

func LoggerInjectMiddleware(baseLogger zerolog.Logger) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {

            reqID := uuid.New().String()

            logger := baseLogger.With().
                Str("request_id", reqID).
                Str("path", r.URL.Path).
                Logger()

            ctx := logger.WithContext(r.Context())

            next.ServeHTTP(w, r.WithContext(ctx))
        })
    }
}

👉 Runs at the start


---

🟡 Middleware 2 (LAST): Request Logging (duration)

func RequestLoggingMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {

        start := time.Now()

        rw := &responseWriter{ResponseWriter: w, status: 200}

        next.ServeHTTP(rw, r)

        duration := time.Since(start)

        logger := zerolog.Ctx(r.Context())

        logger.Info().
            Int("status", rw.status).
            Dur("duration", duration).
            Msg("request completed")
    })
}

👉 Runs at the end

func MetricsMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {

        start := time.Now()

        next.ServeHTTP(w, r)

        duration := time.Since(start).Seconds()

        ctx := r.Context()

        requestCounter.Add(ctx, 1,
            attribute.String("path", r.URL.Path),
        )

        requestLatency.Record(ctx, duration,
            attribute.String("path", r.URL.Path),
        )
    })
}

✅ Minimal Working Example (≈50 lines)

package main

import (
	"context"
	"net/http"
	"os"
	"time"

	"github.com/rs/zerolog"
	"github.com/rs/zerolog/log"

	"go.opentelemetry.io/otel"
	"go.opentelemetry.io/otel/metric"
	"go.opentelemetry.io/contrib/instrumentation/net/http/otelhttp"
)

var (
	meter           = otel.Meter("my-app")
	requestCounter  metric.Int64Counter
	requestLatency  metric.Float64Histogram
)

func main() {
	// logger
	logger := zerolog.New(os.Stdout).With().Timestamp().Str("app", "demo").Logger()

	// metrics init
	requestCounter, _ = meter.Int64Counter("http_requests_total")
	requestLatency, _ = meter.Float64Histogram("http_request_duration")

	// handler
	mux := http.NewServeMux()
	mux.HandleFunc("/hello", func(w http.ResponseWriter, r *http.Request) {
		ctx := r.Context()

		// tracing (child span)
		tr := otel.Tracer("demo")
		ctx, span := tr.Start(ctx, "handler.hello")
		defer span.End()

		// logging (from context)
		l := zerolog.Ctx(ctx)
		l.Info().Msg("handling request")

		time.Sleep(50 * time.Millisecond)
		w.Write([]byte("hello"))
	})

	// middleware: logger + metrics
	handler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()

		// create request logger
		reqLogger := logger.With().
			Str("path", r.URL.Path).
			Logger()

		ctx := reqLogger.WithContext(r.Context())
		r = r.WithContext(ctx)

		mux.ServeHTTP(w, r)

		// metrics
		dur := time.Since(start).Seconds()
		requestCounter.Add(ctx, 1)
		requestLatency.Record(ctx, dur)

		reqLogger.Info().Dur("duration", time.Since(start)).Msg("request done")
	})

	// wrap with otel (tracing)
	finalHandler := otelhttp.NewHandler(handler, "http-server")

	http.ListenAndServe(":8080", finalHandler)
}

❓ Now your important question

> Why not router.Use(otelhttpMiddleware)?




---

🔥 Short answer

> Because otelhttp is a handler wrapper, not a router middleware




---

🧠 Deeper explanation (very important)

In Go HTTP:

Everything is just:

type Handler interface {
    ServeHTTP(w http.ResponseWriter, r *http.Request)
}


---

otelhttp.NewHandler(...) does this:

func NewHandler(h http.Handler, name string) http.Handler

👉 It wraps your handler


---

Think of it like:

your handler → wrapped by otel → returned as new handler


---

Middleware (like router.Use) is different

Routers (chi, gin, echo) define middleware as:

func(next http.Handler) http.Handler


---

🔄 So why not .Use()?

Because:

otelhttp is framework-agnostic

It works with plain net/http

It doesn’t depend on router-specific middleware system



---

✅ But you CAN use it like middleware (if you want)

If your router supports .Use, just wrap it:

func OtelMiddleware(next http.Handler) http.Handler {
    return otelhttp.NewHandler(next, "http-server")
}

Then:

router.Use(OtelMiddleware)

👉 Now it behaves like middleware


---

🎯 When to use which

Approach	When

otelhttp.NewHandler(router, ...)	Simple apps (recommended)
router.Use(OtelMiddleware)	When using middleware-heavy routers
